function doGet(e) { const page = (e && e.parameter.page) || "index"; if (page === "admin") { return HtmlService.createHtmlOutputFromFile("Admin").setTitle("CHIZ Admin").setXFrameOptionsMode(HtmlService.XFrameOptionsMode.ALLOWALL); } else if (page === "projector") { return HtmlService.createHtmlOutputFromFile("projector").setTitle("CHIZ Leaderboard").setXFrameOptionsMode(HtmlService.XFrameOptionsMode.ALLOWALL); } else { // Default – redirect to GitHub page (or serve a landing page) return HtmlService.createHtmlOutput(`

Redirecting to quiz... Click here if not redirected.

`).setXFrameOptionsMode(HtmlService.XFrameOptionsMode.ALLOWALL); } }
